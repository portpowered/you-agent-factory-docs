const result = await agent.run({
  label: "summarizer",
  prompt: "Summarise the request below.\n\n" + args.request,
  modelProvider: args.modelProvider || "",
  model: args.model || "",
});
if (result.status !== "COMPLETED") {
  throw "summarizer failed";
}
return result.output.text.trim();
